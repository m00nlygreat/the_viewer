
class Component extends DCLogic {
componentWillUnmount() {
if (this.mainEl && this._onWheel) {
this.mainEl.removeEventListener('wheel', this._onWheel);
this.mainEl.removeEventListener('gesturestart', this._onGestureStart);
this.mainEl.removeEventListener('gesturechange', this._onGestureChange);
}
if (this._toastT) clearTimeout(this._toastT);
}

uid() { this._id = (this._id || 100) + 1; return 'n' + this._id; }

initModel() {
const V = (vtype, text) => ({ id: this.uid(), t: 'val', vtype: vtype, text: text });
const S = (key, node) => ({ id: this.uid(), key: key, node: node });
const O = (slots, collapsed) => ({ id: this.uid(), t: 'obj', items: slots, collapsed: !!collapsed });
const A = (slots) => ({ id: this.uid(), t: 'arr', items: slots, collapsed: false });
const doc = O([
S('id', V('text', 'A-1042')),
S('customer', O([S('name', V('text', '한서연')), S('tier', V('text', 'gold'))])),
S('recipient', O([S('name', V('text', '이도윤'))], true)),
S('total', null),
S('paid', V('bool', 'true')),
S('wrapColor', V('color', '#E11D48')),
S('placedAt', V('date', '2026-10-27T09:12:44+09:00')),
S('tags', A([S(null, V('text', 'gift')), S(null, V('text', 'express')), S(null, null)])),
S('note', V('null', ''))
]);
return [
{ id: 'doc', x: 48, y: 40, isDoc: true, node: doc },
{ id: this.uid(), x: 620, y: 620, slot: S('memo', V('text', '확인 필요')) }
];
}

getRoots() {
const s = this.state || {};
if (s.roots) return s.roots;
if (!this._initRoots) this._initRoots = this.initModel();
return this._initRoots;
}

locNode(node, id, info) {
if (node.id === id) return Object.assign({ kind: 'node', node: node }, info);
if (node.items) {
for (let i = 0; i < node.items.length; i++) {
const r = this.locSlot(node.items[i], id, { parent: node, index: i });
if (r) return r;
}
}
return null;
}
locSlot(slot, id, info) {
if (slot.id === id) return Object.assign({ kind: 'slot', slot: slot }, info);
if (slot.node) return this.locNode(slot.node, id, { holder: 'slot', slot: slot });
return null;
}
locate(R, id) {
for (let i = 0; i < R.length; i++) {
const r = R[i];
if (r.id === id) return { kind: 'root', root: r };
const res = r.node ? this.locNode(r.node, id, { holder: 'root', root: r }) : this.locSlot(r.slot, id, { holder: 'rootslot', root: r });
if (res) return res;
}
return null;
}
collectIds(x, set) {
if (!x) return;
set.add(x.id);
if (x.node !== undefined && x.t === undefined) { this.collectIds(x.node, set); return; }
if (x.items) x.items.forEach((s) => this.collectIds(s, set));
}
uniqueKey(c, base) {
const keys = new Set(c.items.map((s) => s.key));
if (!keys.has(base)) return base;
let i = 2;
while (keys.has(base + i)) i++;
return base + i;
}
detach(R, src) {
if (src.type === 'root') {
const i = R.findIndex((r) => r.id === src.id);
if (i < 0 || R[i].isDoc) return null;
const r = R.splice(i, 1)[0];
return r.node ? { node: r.node } : { slot: r.slot };
}
const l = this.locate(R, src.id);
if (!l) return null;
if (l.kind === 'slot') {
if (l.parent) l.parent.items.splice(l.index, 1);
else if (l.holder === 'rootslot') R.splice(R.indexOf(l.root), 1);
return { slot: l.slot };
}
if (l.kind === 'node') {
if (l.holder === 'slot') l.slot.node = null;
else if (l.holder === 'root') { if (l.root.isDoc) return null; R.splice(R.indexOf(l.root), 1); }
return { node: l.node };
}
return null;
}
pushHist() {
this._hist = this._hist || [];
this._hist.push(JSON.stringify(this.getRoots()));
if (this._hist.length > 100) this._hist.shift();
this._redo = [];
}
mutate(fn, record) {
const R = JSON.parse(JSON.stringify(this.getRoots()));
if (record !== false) this.pushHist();
fn(R);
this.setState({ roots: R });
}
undo() {
const h = this._hist || [];
if (!h.length) return;
(this._redo = this._redo || []).push(JSON.stringify(this.getRoots()));
this.setState({ roots: JSON.parse(h.pop()), edit: null, datePop: null, sel: null });
}
redo() {
const r = this._redo || [];
if (!r.length) return;
(this._hist = this._hist || []).push(JSON.stringify(this.getRoots()));
this.setState({ roots: JSON.parse(r.pop()), edit: null, datePop: null, sel: null });
}
toast(msg) {
if (this._toastT) clearTimeout(this._toastT);
this.setState({ toast: msg });
this._toastT = setTimeout(() => this.setState({ toast: null }), 1800);
}
toWorld(el) {
const r = el.getBoundingClientRect(), m = this.mainEl.getBoundingClientRect();
const s = this.state || {};
const z = s.zoom || 1;
return { x: (r.left - m.left - (s.panX || 0)) / z, y: (r.top - m.top - (s.panY || 0)) / z, w: r.width / z, h: r.height / z };
}
zoomTo(target, cx, cy) {
const s = this.state || {};
const z = s.zoom || 1, px = s.panX || 0, py = s.panY || 0;
const nz = Math.min(3, Math.max(0.25, target));
if (cx == null) {
const r = this.mainEl ? this.mainEl.getBoundingClientRect() : { width: 800, height: 600 };
cx = r.width / 2; cy = r.height / 2;
}
this.setState({ zoom: nz, panX: cx - (cx - px) * nz / z, panY: cy - (cy - py) * nz / z });
}

parseDate(text) {
const m = (text || '').match(/^(\d{4}-\d\d-\d\d)?T?(\d\d:\d\d(?::\d\d)?)?(Z|[+-]\d\d:\d\d)?$/) || [];
return { d: !!m[1] || !m[2], t: !!m[2], z: !!m[3], dv: m[1] || '2026-10-29', tv: m[2] || '09:00:00', zv: m[3] || '+09:00' };
}
composeDate(p) {
const zEff = p.t && p.z;
if (p.d && p.t) return p.dv + 'T' + p.tv + (zEff ? p.zv : '');
if (p.d) return p.dv;
return p.tv + (zEff ? p.zv : '');
}
updateDatePop(patch) {
const s = this.state || {};
const dp = Object.assign({}, s.datePop, patch);
if (!dp.d && !dp.t) return;
const iso = this.composeDate(dp);
this.mutate((R) => { const l = this.locate(R, dp.id); if (l && l.node) l.node.text = iso; }, false);
this.setState({ datePop: dp });
}

validate(vtype, v) {
if (vtype === 'number') return /^-?(0|[1-9]\d*)(\.\d+)?([eE][+-]?\d+)?$/.test(v.trim()) ? '' : '숫자가 아니에요';
if (vtype === 'color') return /^#([0-9a-fA-F]{6}|[0-9a-fA-F]{8})$/.test(v.trim()) || /^(rgb|rgba|hsl|hsla|oklch|oklab|lab|lch)\(.+\)$/.test(v.trim()) ? '' : '#RRGGBB 형식으로 입력하세요';
return '';
}
beginEdit(id, el) {
const l = this.locate(this.getRoots(), id);
if (!l || !l.node || l.node.t !== 'val' || !this.mainEl) return;
const n = l.node;
const w = this.toWorld(el);
if (n.vtype === 'bool') {
this.mutate((R) => { const m = this.locate(R, id); m.node.text = m.node.text === 'true' ? 'false' : 'true'; });
this.setState({ sel: id, edit: null, datePop: null });
return;
}
if (n.vtype === 'null') { this.setState({ sel: id }); return; }
if (n.vtype === 'date') {
this.pushHist();
this.setState({ sel: id, edit: null, datePop: Object.assign({ id: id, x: w.x, y: w.y + w.h + 6 }, this.parseDate(n.text)) });
return;
}
this.setState({ sel: id, datePop: null, edit: { kind: 'val', id: id, x: w.x, y: w.y, w: Math.max(w.w + 24, 120), h: Math.max(w.h, 26), value: n.text, err: '', vtype: n.vtype } });
}
beginKeyEdit(slotId, el) {
const l = this.locate(this.getRoots(), slotId);
if (!l || !l.slot || !this.mainEl) return;
const w = this.toWorld(el);
this.setState({ sel: slotId, datePop: null, edit: { kind: 'key', id: slotId, x: w.x - 6, y: w.y - 4, w: Math.max(w.w + 24, 100), h: Math.max(w.h + 8, 26), value: l.slot.key || '', err: '', vtype: 'text' } });
}
commitEdit() {
const s = this.state || {};
const ed = s.edit;
if (!ed) return true;
if (ed.kind === 'key') {
const t = ed.value.trim();
if (!t) { this.setState({ edit: Object.assign({}, ed, { err: '이름을 입력하세요' }) }); return false; }
this.mutate((R) => { const l = this.locate(R, ed.id); if (l && l.slot) l.slot.key = t; });
this.setState({ edit: null });
return true;
}
const err = this.validate(ed.vtype, ed.value);
if (err) { this.setState({ edit: Object.assign({}, ed, { err: err }) }); return false; }
const val = ed.vtype === 'text' ? ed.value : ed.value.trim();
this.mutate((R) => { const l = this.locate(R, ed.id); if (l && l.node) l.node.text = val; });
this.setState({ edit: null });
return true;
}
deleteSel() {
const s = this.state || {};
if (!s.sel) return;
const l = this.locate(this.getRoots(), s.sel);
if (!l) return;
if (l.kind === 'node' && l.holder === 'root' && l.root.isDoc) return;
this.mutate((R) => { this.detach(R, { type: l.kind === 'slot' ? 'slot' : 'node', id: s.sel }); });
this.setState({ sel: null, edit: null, datePop: null });
}

startDrag(e, src, selId) {
const s = this.state || {};
if (s.space || e.button !== 0) return;
e.stopPropagation();
e.preventDefault();
try { e.currentTarget.setPointerCapture(e.pointerId); } catch (err) {}
const r = e.currentTarget.getBoundingClientRect();
let preview, kind;
const ids = new Set();
if (src.type === 'palette') {
preview = src.make();
} else {
const l = this.locate(this.getRoots(), src.id);
if (!l) return;
if (src.type === 'slot') preview = { slot: l.slot };
else if (src.type === 'node') preview = { node: l.node };
else preview = l.root.node ? { node: l.root.node } : { slot: l.root.slot };
this.collectIds(preview.slot || preview.node, ids);
}
kind = preview.slot ? 'slot' : 'node';
this._drag = { src: src, selId: selId, preview: preview, kind: kind, ids: ids, sx: e.clientX, sy: e.clientY, ox: e.clientX - r.left, oy: e.clientY - r.top, moved: false };
}
findTarget(x, y) {
const el = document.elementFromPoint(x, y);
if (!el || !this.mainEl || !this.mainEl.contains(el)) return null;
const d = this._drag;
if (d.kind === 'node') {
const sk = el.closest('[data-sock]');
if (sk) { const sid = sk.getAttribute('data-sock'); if (!d.ids.has(sid)) return 'sk:' + sid; }
}
let cont = el.closest('[data-cont]');
while (cont) {
const cid = cont.getAttribute('data-cont');
if (!d.ids.has(cid)) {
if (cont.getAttribute('data-collapsed') === 'true') return 'ins:' + cid + ':' + cont.getAttribute('data-len');
const rows = Array.from(cont.querySelectorAll('[data-parent="' + cid + '"]'));
let idx = rows.length;
for (let i = 0; i < rows.length; i++) {
const rr = rows[i].getBoundingClientRect();
if (y < rr.top + rr.height / 2) { idx = i; break; }
}
return 'ins:' + cid + ':' + idx;
}
cont = cont.parentElement ? cont.parentElement.closest('[data-cont]') : null;
}
return 'canvas';
}
applyDrop(d, target, e) {
const s = this.state || {};
const m = this.mainEl.getBoundingClientRect();
const z = s.zoom || 1;
const wx = Math.round((e.clientX - d.ox - m.left - (s.panX || 0)) / z);
const wy = Math.round((e.clientY - d.oy - m.top - (s.panY || 0)) / z);
this.mutate((R) => {
if (target === 'canvas' && d.src.type === 'root') {
const rt = R.find((x) => x.id === d.src.id);
if (rt) { rt.x = wx; rt.y = wy; }
return;
}
let adj = 0;
if (target.indexOf('ins:') === 0 && d.src.type === 'slot') {
const p = target.split(':');
const l = this.locate(R, d.src.id);
if (l && l.parent && l.parent.id === p[1] && l.index < Number(p[2])) adj = -1;
}
const pl = d.src.type === 'palette' ? d.src.make() : this.detach(R, d.src);
if (!pl) return;
if (target === 'canvas') {
R.push(Object.assign({ id: this.uid(), x: wx, y: wy }, pl.slot ? { slot: pl.slot } : { node: pl.node }));
return;
}
if (target.indexOf('sk:') === 0) {
const l = this.locate(R, target.slice(3));
if (l && l.slot) l.slot.node = pl.node || (pl.slot ? pl.slot.node : null);
return;
}
if (target.indexOf('ins:') === 0) {
const p = target.split(':');
const l = this.locate(R, p[1]);
if (!l || !l.node || !l.node.items) return;
const c = l.node;
let slot;
if (pl.slot) {
slot = pl.slot;
if (c.t === 'obj' && (slot.key == null || slot.key === '')) slot.key = this.uniqueKey(c, 'key');
if (c.t === 'arr') slot.key = null;
} else {
slot = { id: this.uid(), key: c.t === 'obj' ? this.uniqueKey(c, 'key') : null, node: pl.node };
}
c.items.splice(Math.max(0, Math.min(c.items.length, Number(p[2]) + adj)), 0, slot);
c.collapsed = false;
}
});
this.setState({ dragging: false, hover: null, sel: d.selId || null });
}

fromJson(v) {
if (Array.isArray(v)) return { id: this.uid(), t: 'arr', collapsed: false, items: v.map((x) => ({ id: this.uid(), key: null, node: this.fromJson(x) })) };
if (v !== null && typeof v === 'object') return { id: this.uid(), t: 'obj', collapsed: false, items: Object.keys(v).map((k) => ({ id: this.uid(), key: k, node: this.fromJson(v[k]) })) };
if (v === null) return { id: this.uid(), t: 'val', vtype: 'null', text: '' };
if (typeof v === 'boolean') return { id: this.uid(), t: 'val', vtype: 'bool', text: String(v) };
if (typeof v === 'number') return { id: this.uid(), t: 'val', vtype: 'number', text: String(v) };
const sv = String(v);
let vt = 'text';
if (/^\d{4}-(0[1-9]|1[0-2])-(0[1-9]|[12]\d|3[01])(T([01]\d|2[0-3]):[0-5]\d(:[0-5]\d(\.\d+)?)?(Z|[+-]\d\d:\d\d)?)?$/.test(sv)) vt = 'date';
else if (/^#([0-9a-fA-F]{6}|[0-9a-fA-F]{8})$/.test(sv)) vt = 'color';
return { id: this.uid(), t: 'val', vtype: vt, text: sv };
}

renderVals() {
const st = this.state || {};
if (!this._onWheel) {
this._onWheel = (e) => {
e.preventDefault();
const s = this.state || {};
if (e.ctrlKey || e.metaKey) {
const r = this.mainEl.getBoundingClientRect();
this.zoomTo((s.zoom || 1) * Math.exp(-e.deltaY * 0.01), e.clientX - r.left, e.clientY - r.top);
} else {
this.setState({ panX: (s.panX || 0) - e.deltaX, panY: (s.panY || 0) - e.deltaY });
}
};
this._onGestureStart = (e) => { e.preventDefault(); this._gz = (this.state || {}).zoom || 1; };
this._onGestureChange = (e) => {
e.preventDefault();
const r = this.mainEl.getBoundingClientRect();
this.zoomTo((this._gz || 1) * e.scale, e.clientX - r.left, e.clientY - r.top);
};
this._mainRef = (el) => {
if (el === this.mainEl) return;
if (this.mainEl) {
this.mainEl.removeEventListener('wheel', this._onWheel);
this.mainEl.removeEventListener('gesturestart', this._onGestureStart);
this.mainEl.removeEventListener('gesturechange', this._onGestureChange);
}
this.mainEl = el;
if (el) {
el.addEventListener('wheel', this._onWheel, { passive: false });
el.addEventListener('gesturestart', this._onGestureStart, { passive: false });
el.addEventListener('gesturechange', this._onGestureChange, { passive: false });
}
};
this._worldRef = (el) => { this.worldEl = el; };
this._edRef = (el) => {
if (el && el !== this._edEl) { this._edEl = el; el.focus(); el.select(); }
if (!el) this._edEl = null;
};
}
const noop = () => {};
const zoom = st.zoom || 1, panX = st.panX || 0, panY = st.panY || 0;
const leftW = st.leftW || 220, rightW = st.rightW || 340;
const clampL = (w) => Math.min(360, Math.max(180, w));
const clampR = (w) => Math.min(560, Math.max(260, w));
const isTyping = (t) => t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable);

const viewVals = {
mainRef: this._mainRef,
worldRef: this._worldRef,
edRef: this._edRef,
stop: (e) => e.stopPropagation(),
worldTf: 'translate(' + panX + 'px, ' + panY + 'px) scale(' + zoom + ')',
dotSize: (20 * zoom) + 'px ' + (20 * zoom) + 'px',
dotPos: panX + 'px ' + panY + 'px',
cursor: st.panning ? 'grabbing' : (st.space ? 'grab' : 'default'),
zoomPct: Math.round(zoom * 100) + '%',
zoomIn: () => this.zoomTo(zoom * 1.2),
zoomOut: () => this.zoomTo(zoom / 1.2),
zoomReset: () => this.setState({ zoom: 1, panX: 0, panY: 0 }),
fit: () => {
if (!this.mainEl || !this.worldEl) return;
const r = this.mainEl.getBoundingClientRect();
const ww = this.worldEl.offsetWidth, wh = this.worldEl.offsetHeight;
const nz = Math.min(3, Math.max(0.25, Math.min(r.width / ww, r.height / wh)));
this.setState({ zoom: nz, panX: (r.width - ww * nz) / 2, panY: (r.height - wh * nz) / 2 });
},
mainEnter: () => {
const a = document.activeElement;
if (!isTyping(a) && this.mainEl && a !== this.mainEl) this.mainEl.focus({ preventScroll: true });
},
mainKeyDown: (e) => {
if (isTyping(e.target)) return;
const mod = e.metaKey || e.ctrlKey;
if (e.code === 'Space' || e.key === ' ') { e.preventDefault(); if (!st.space) this.setState({ space: true }); return; }
if (mod && (e.key === 'z' || e.key === 'Z')) { e.preventDefault(); if (e.shiftKey) this.redo(); else this.undo(); return; }
if (mod && (e.key === 'y' || e.key === 'Y')) { e.preventDefault(); this.redo(); return; }
if ((e.key === 'Delete' || e.key === 'Backspace') && st.sel) { e.preventDefault(); this.deleteSel(); return; }
if (e.key === 'Escape') { this.setState({ sel: null, edit: null, datePop: null }); return; }
if (e.key === 'Enter' && st.sel) {
e.preventDefault();
const l = this.locate(this.getRoots(), st.sel);
if (!l) return;
if (l.kind === 'node' && l.node.t === 'val') { const el = document.querySelector('[data-nid="' + st.sel + '"]'); if (el) this.beginEdit(st.sel, el); }
if (l.kind === 'slot') { const row = document.querySelector('[data-row="' + st.sel + '"]'); const k = row && row.querySelector('[data-key]'); if (k) this.beginKeyEdit(st.sel, k); }
}
},
mainKeyUp: (e) => { if (e.code === 'Space' || e.key === ' ') this.setState({ space: false }); },
mainBlur: () => { if (st.space) this.setState({ space: false }); },
panDown: (e) => {
if (e.pointerType === 'touch') {
if (!this._ptrs) this._ptrs = new Map();
this._ptrs.set(e.pointerId, { x: e.clientX, y: e.clientY });
if (this._ptrs.size === 2) {
const [a, b] = Array.from(this._ptrs.values());
const r = this.mainEl.getBoundingClientRect();
this._pan = null;
this._drag = null;
this._pinch = { d: Math.hypot(a.x - b.x, a.y - b.y) || 1, mx: (a.x + b.x) / 2 - r.left, my: (a.y + b.y) / 2 - r.top, z: zoom, px: panX, py: panY };
this.setState({ panning: false, dragging: false, hover: null });
try { e.currentTarget.setPointerCapture(e.pointerId); } catch (err) {}
return;
}
}
const onBg = e.target === this.mainEl || e.target === this.worldEl;
if (!(e.button === 1 || (e.button === 0 && (onBg || st.space)))) return;
e.preventDefault();
try { e.currentTarget.setPointerCapture(e.pointerId); } catch (err) {}
this._pan = { x: e.clientX, y: e.clientY, px: panX, py: panY };
const patch = { panning: true };
if (onBg && !st.space) { patch.sel = null; patch.datePop = null; }
this.setState(patch);
},
panMove: (e) => {
if (this._ptrs && this._ptrs.has(e.pointerId)) this._ptrs.set(e.pointerId, { x: e.clientX, y: e.clientY });
if (this._pinch && this._ptrs && this._ptrs.size >= 2) {
const [a, b] = Array.from(this._ptrs.values());
const r = this.mainEl.getBoundingClientRect();
const p = this._pinch;
const nz = Math.min(3, Math.max(0.25, p.z * Math.hypot(a.x - b.x, a.y - b.y) / p.d));
const mx = (a.x + b.x) / 2 - r.left, my = (a.y + b.y) / 2 - r.top;
const wx = (p.mx - p.px) / p.z, wy = (p.my - p.py) / p.z;
this.setState({ zoom: nz, panX: mx - wx * nz, panY: my - wy * nz });
return;
}
if (!this._pan) return;
this.setState({ panX: this._pan.px + e.clientX - this._pan.x, panY: this._pan.py + e.clientY - this._pan.y });
},
panEnd: (e) => {
if (this._ptrs && e) this._ptrs.delete(e.pointerId);
if (this._pinch && (!this._ptrs || this._ptrs.size < 2)) this._pinch = null;
if (this._pan) { this._pan = null; this.setState({ panning: false }); }
},
leftW: leftW,
rightW: rightW,
lActive: st.rz === 'l' ? 'true' : 'false',

} : { x: 0, y: 0, d: false, t: false, zOn: false, dv: '', tv: '', zv: '', dOff: true, tOff: true, zOff: true, zLocked: true, dFg: '#111111', tFg: '#111111', zFg: '#111111', zLabelFg: '#404040', iso: '' };

const makeFor = (src) => {
if (src === 'obj' || src === 'arr') return () => ({ node: { id: this.uid(), t: src, items: [], collapsed: false } });
if (src === 'prop') return () => ({ slot: { id: this.uid(), key: 'key', node: null } });
if (src === 'idx') return () => ({ slot: { id: this.uid(), key: null, node: null } });
if (src.indexOf('v:') === 0) { const t = src.slice(2); return () => ({ node: { id: this.uid(), t: 'val', vtype: t, text: VT[t].def } }); }
if (src.indexOf('schema:') === 0) {
const def = (this._schemaDefs || [])[Number(src.slice(7))];
if (!def) return null;
return () => ({ node: { id: this.uid(), t: 'obj', collapsed: false, items: def.fields.map((f) => ({ id: this.uid(), key: f.key, node: null })) } });
}
return null;
};

const dndVals = {
roots: rootViews,
palVals: palVals,
schemas: schemaDefs.map((d, i) => ({ src: 'schema:' + i, name: d.name, count: d.count, fields: d.fields })),
schemaCount: schemaDefs.length,
dragging: dragging,
gx: st.gx || 0, gy: st.gy || 0,
ghostIsValue: ghostIsValue,
ghostIsBlock: !!gp && !ghostIsValue,
ghostV: ghostIsValue ? viewVal(gNode, {}) : EMPTY_V,
ghostLabel: ghostLabel,
ghostBg: gp && gp.slot ? '#F5F5F5' : '#FFFFFF',
srcDown: (e) => {
const src = e.currentTarget.getAttribute('data-src');
const make = makeFor(src || '');
if (!make) return;
this.startDrag(e, { type: 'palette', make: make }, null);
},
dragMove: (e) => {
const dd = this._drag;
if (!dd) return;
if (!dd.moved && Math.hypot(e.clientX - dd.sx, e.clientY - dd.sy) < 4) return;
dd.moved = true;
const target = this.findTarget(e.clientX, e.clientY);
this.setState({ dragging: true, gx: e.clientX - dd.ox, gy: e.clientY - dd.oy, hover: target, edit: null, datePop: null });
},
dragEnd: (e) => {
const dd = this._drag;
if (!dd) return;
this._drag = null;
if (!dd.moved) { if (dd.selId) this.setState({ sel: dd.selId, datePop: null }); return; }
const target = this.findTarget(e.clientX, e.clientY);
if (!target) { this.setState({ dragging: false, hover: null }); return; }
this.applyDrop(dd, target, e);
},
dragCancel: () => { this._drag = null; this.setState({ dragging: false, hover: null }); },
editing: !!ed,
ed: edView,
edChange: (e) => this.setState({ edit: Object.assign({}, st.edit, { value: e.target.value, err: '' }) }),
edColor: (e) => this.setState({ edit: Object.assign({}, st.edit, { value: e.target.value.toUpperCase(), err: '' }) }),
edKey: (e) => {
e.stopPropagation();
if (e.key === 'Enter') { e.preventDefault(); this.commitEdit(); }
if (e.key === 'Escape') { e.preventDefault(); this.setState({ edit: null }); if (this.mainEl) this.mainEl.focus({ preventScroll: true }); }
},
edBlur: () => { if ((this.state || {}).edit && !this.commitEdit()) this.setState({ edit: null }); },
dpOpen: !!dpS,
dp: dpView,
dpToggleD: () => this.updateDatePop({ d: !dpS.d }),
dpToggleT: () => this.updateDatePop({ t: !dpS.t }),
dpToggleZ: () => this.updateDatePop({ z: !dpS.z }),
dpDv: (e) => this.updateDatePop({ dv: e.target.value }),
dpTv: (e) => this.updateDatePop({ tv: e.target.value }),
dpZv: (e) => this.updateDatePop({ zv: e.target.value }),
outLines: lines,
tomlNoteShow: fmt === 'toml' && dropped.length > 0,
tomlNote: 'TOML에는 \'없음\' 값이 없어서 ' + dropped.map((p) => p.replace(/^\./, '')).join(', ') + '은(는) 빠졌어요.',
checks: checks,
isJson: fmt === 'json', isYaml: fmt === 'yaml', isToml: fmt === 'toml',
jsonFg: fmt === 'json' ? '#111111' : '#737373', yamlFg: fmt === 'yaml' ? '#111111' : '#737373', tomlFg: fmt === 'toml' ? '#111111' : '#737373',
jsonLine: fmt === 'json' ? 'inset 0 -2px 0 #111111' : 'none', yamlLine: fmt === 'yaml' ? 'inset 0 -2px 0 #111111' : 'none', tomlLine: fmt === 'toml' ? 'inset 0 -2px 0 #111111' : 'none',
showJson: () => this.setState({ fmt: 'json' }),
showYaml: () => this.setState({ fmt: 'yaml' }),
showToml: () => this.setState({ fmt: 'toml' }),
hasToast: !!st.toast,
toast: st.toast || '',
exportCopy: () => {
const label = fmt.toUpperCase();
try {
navigator.clipboard.writeText(plain).then(() => this.toast(label + '을 클립보드에 복사했어요'), () => this.toast('클립보드에 복사할 수 없어요'));
} catch (err) { this.toast('클립보드에 복사할 수 없어요'); }
},
importOpen: !!st.importOpen,
importText: st.importText || '',
importErr: st.importErr || '',
importHasErr: !!st.importErr,
openImport: () => this.setState({ importOpen: true, importErr: '' }),
closeImport: () => this.setState({ importOpen: false, importErr: '' }),
importChange: (e) => this.setState({ importText: e.target.value, importErr: '' }),
doImport: () => {
let v;
try { v = JSON.parse(st.importText || ''); } catch (err) { this.setState({ importErr: 'JSON을 읽을 수 없어요: ' + err.message }); return; }
if (v === null || typeof v !== 'object') { this.setState({ importErr: '최상위는 객체나 배열이어야 해요' }); return; }
const node = this.fromJson(v);
this.mutate((R) => { const d = R.find((r) => r.isDoc); if (d) d.node = node; });
this.setState({ importOpen: false, importErr: '', focus: null, sel: null, edit: null, datePop: null });
this.toast('불러왔어요');
}
};

return Object.assign({}, viewVals, dndVals);
}
